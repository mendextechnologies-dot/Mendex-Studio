import sys,os,json,http.server,threading; from bridge import Bridge
ROOT,DIST,PORT=sys.argv[1],sys.argv[2],int(sys.argv[3]); b=Bridge(ROOT); lk=threading.Lock()
class H(http.server.SimpleHTTPRequestHandler):
    def __init__(s,*a,**k): super().__init__(*a,directory=DIST,**k)
    def do_POST(s):
        body=json.loads(s.rfile.read(int(s.headers["Content-Length"])))
        i,t=b.send(body["method"],body.get("params")); d,_=b.wait(i,t); d["id"]=body["id"]
        out=json.dumps(d).encode(); s.send_response(200); s.send_header("Content-Type","application/json"); s.end_headers(); s.wfile.write(out)
    def log_message(s,*a): pass
http.server.ThreadingHTTPServer(("127.0.0.1",PORT),H).serve_forever()
