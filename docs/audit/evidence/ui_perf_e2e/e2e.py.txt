R={}
def log(step,status,ev): R[step]=(status,ev); print(f"## {step}: {status}\n{ev}\n",flush=True)
srv=MendexMCPServer(ROOT)
def mcp(name,args):
    r=srv.handle_request({"jsonrpc":"2.0","id":1,"method":"tools/call","params":{"name":name,"arguments":args}})["result"]
    t=r["content"][0]["text"]
    try: return r["isError"], json.loads(t)
    except Exception: return r["isError"], {"raw":t[:600]}
def cli(*a,req=None):
    env=dict(os.environ,PYTHONPATH=REPO+r"\src")
    args=[PY,"-m","mendex_core.cli","--root",ROOT,*a]
    if req is not None:
        p=os.path.join(ROOT,"..",f"req-{time.time_ns()}.json"); open(p,"w").write(json.dumps(req)); args.append(p)
    c=subprocess.run(args,capture_output=True,text=True,cwd=REPO,env=env,timeout=600); 
    try: return c.returncode,json.loads(c.stdout or c.stderr)
    except Exception: return c.returncode,(c.stdout+c.stderr)[-800:]
